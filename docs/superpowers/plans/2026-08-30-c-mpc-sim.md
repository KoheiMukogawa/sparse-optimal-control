**Goal:** 経路追従のスパース最適制御（Maximum Hands-off / L1-MPC）を、外部ライブラリに依存しない C99 で実装し、Python 版と数値一致することを検証したうえで、入力遅延下の L1 チャタリングとその抑制を再現する。

**Tech Stack:** C99 のみ（標準ライブラリと `libm` 以外の依存なし）、GNU Make、参照データ生成にのみ Python + cvxpy（卒研リポジトリ側）

## 定式化（全タスク共通の前提）

- 状態 `z = (x_e, y_e, θ_e)`, `nx = 3`。入力 `δu = (δv, δω)`, `nu = 2`
- `A = I + ts·A_c`,  `A_c = [[0, w_r, 0], [−w_r, 0, v_r], [0, 0, 0]]`
- `B = ts·B_c`,  `B_c = [[−1, 0], [0, 0], [0, −1]]`
- A, B はホライズン内で一定（`v_r, w_r` を現在値で固定）
- コスト:
  `J = Σ_{k=0}^{N−1} [ zₖᵀQzₖ + reg(δuₖ) + ms‖δuₖ − δu_{k−1}‖² ] + z_Nᵀ Q_f z_N`
  - `reg(δu) = δuᵀRδu`（L2）/ `λ‖δu‖₁`（L1）
  - `k=0` の `δu_{−1}` は前ステップに実適用した `δu`
- 制約: `z_{k+1} = Azₖ + Bδuₖ`, `|u_nom + δuₖ| ≤ u_max`（要素ごと）
- 既定値: `N=15, ts=0.1, Q=diag(10,10,1), R=diag(1,1), qf_scale=10, v_max=0.15, w_max=2.0`

### Condensing の式（Task 6 で実装、ここが全体の要）

`U = [δu₀; …; δu_{N−1}]` として `Z = Φz₀ + ΓU`。

- `Φ` は `3(N+1) × 3`、ブロック行 `k` が `A^k`
- `Γ` は `3(N+1) × 2N`、ブロック `(k, j)` が `j < k` のとき `A^{k−1−j}B`、それ以外 0
- `Q̃ = blkdiag(0, Q, …, Q, Q_f)`（ブロック 0 が 0 なのは `z₀` が定数のため）
- `M = DᵀD`: ブロック対角が `k ≤ N−2` で `2I`、`k = N−1` で `I`、隣接ブロック `(k,k−1)` と `(k−1,k)` が `−I`

これらから

```
P  = ΓᵀQ̃Γ + R̃ + ms·M          （L1 では R̃ = 0）
q  = ΓᵀQ̃Φz₀ − ms·[du_prev; 0; …; 0]
c0 = z₀ᵀQz₀ + (Φz₀)ᵀQ̃(Φz₀) + ms‖du_prev‖²
J  = UᵀPU + 2qᵀU + c0 + λ‖U‖₁
```

`c0` は Python の `prob.value` と突き合わせるために必須（Task 9）。

```c
#define TEST_UTIL_H

#include <stdio.h>

static int tu_checks = 0;
static int tu_failures = 0;

#define TU_CHECK(cond)                                                      \
    do {                                                                    \
        tu_checks++;                                                        \
        if (!(cond)) {                                                      \
            tu_failures++;                                                  \
            fprintf(stderr, "FAIL %s:%d: %s\n", __FILE__, __LINE__, #cond); \
        }                                                                   \
    } while (0)

#define TU_NEAR(a, b, tol)                                                   \
    do {                                                                     \
        double tu_a = (a);                                                   \
        double tu_b = (b);                                                   \
        tu_checks++;                                                         \
        if (!(fabs(tu_a - tu_b) <= (tol))) {                                 \
            tu_failures++;                                                   \
            fprintf(stderr, "FAIL %s:%d: %s=%.9g != %s=%.9g (tol=%.3g)\n",   \
                    __FILE__, __LINE__, #a, tu_a, #b, tu_b, (double)(tol));  \
        }                                                                    \
    } while (0)

#define TU_REPORT()                                                         \
    do {                                                                    \
        printf("%-28s %3d checks, %d failures\n",                           \
               __FILE__, tu_checks, tu_failures);                           \
        return tu_failures == 0 ? 0 : 1;                                    \
    } while (0)

#endif /* TEST_UTIL_H */
```

```c
#include "linalg.h"

#include <math.h>

void la_zero(double *v, int n)
{
    int i;
    for (i = 0; i < n; i++) {
        v[i] = 0.0;
    }
}

void la_copy(double *dst, const double *src, int n)
{
    int i;
    for (i = 0; i < n; i++) {
        dst[i] = src[i];
    }
}

double la_dot(const double *a, const double *b, int n)
{
    double s = 0.0;
    int i;
    for (i = 0; i < n; i++) {
        s += a[i] * b[i];
    }
    return s;
}

void la_axpy(double alpha, const double *x, double *y, int n)
{
    int i;
    for (i = 0; i < n; i++) {
        y[i] += alpha * x[i];
    }
}

void la_scale(double *v, double alpha, int n)
{
    int i;
    for (i = 0; i < n; i++) {
        v[i] *= alpha;
    }
}

double la_norm_inf(const double *v, int n)
{
    double m = 0.0;
    int i;
    for (i = 0; i < n; i++) {
        double a = fabs(v[i]);
        if (a > m) {
            m = a;
        }
    }
    return m;
}

void la_matvec(const double *A, const double *x, double *y, int m, int n)
{
    int i, j;
    for (i = 0; i < m; i++) {
        double s = 0.0;
        for (j = 0; j < n; j++) {
            s += A[i * n + j] * x[j];
        }
        y[i] = s;
    }
}

void la_matvec_t(const double *A, const double *x, double *y, int m, int n)
{
    int i, j;
    la_zero(y, n);
    for (i = 0; i < m; i++) {
        double xi = x[i];
        if (xi == 0.0) {
            continue;
        }
        for (j = 0; j < n; j++) {
            y[j] += A[i * n + j] * xi;
        }
    }
}

void la_matmul(const double *A, const double *B, double *C, int m, int k, int n)
{
    int i, p, j;
    for (i = 0; i < m; i++) {
        for (j = 0; j < n; j++) {
            C[i * n + j] = 0.0;
        }
        for (p = 0; p < k; p++) {
            double a = A[i * k + p];
            if (a == 0.0) {
                continue;
            }
            for (j = 0; j < n; j++) {
                C[i * n + j] += a * B[p * n + j];
            }
        }
    }
}
```

```c
#include "path.h"

#include <math.h>

int path_init(path_t *p, const double *xs, const double *ys, int n)
{
    int i;
    if (n < 2 || n > PATH_WP_MAX) {
        return -1;
    }
    for (i = 0; i < n; i++) {
        p->x[i] = xs[i];
        p->y[i] = ys[i];
    }
    p->n = n;
    return 0;
}

double path_normalize_angle(double a)
{
    return atan2(sin(a), cos(a));
}

void path_reference_pose(const path_t *p, double x, double y, double lookahead,
                         double *x_r, double *y_r, double *th_r)
{
    double best_d2 = -1.0;
    int best_i = 0;
    double best_t = 0.0;
    double remain;
    int i;

    /* 1) 全セグメントへの射影のうち最近傍を選ぶ */
    for (i = 0; i < p->n - 1; i++) {
        double ax = p->x[i], ay = p->y[i];
        double dx = p->x[i + 1] - ax, dy = p->y[i + 1] - ay;
        double seg_len2 = dx * dx + dy * dy;
        double t, px, py, d2;

        if (seg_len2 <= 0.0) {
            continue;   /* 退化セグメントは無視する */
        }
        t = ((x - ax) * dx + (y - ay) * dy) / seg_len2;
        if (t < 0.0) {
            t = 0.0;
        } else if (t > 1.0) {
            t = 1.0;
        }
        px = ax + t * dx;
        py = ay + t * dy;
        d2 = (x - px) * (x - px) + (y - py) * (y - py);
        if (best_d2 < 0.0 || d2 < best_d2) {
            best_d2 = d2;
            best_i = i;
            best_t = t;
        }
    }

    /* 2) 射影点から経路に沿って lookahead だけ進む */
    remain = lookahead;
    i = best_i;
    while (i < p->n - 1) {
        double ax = p->x[i], ay = p->y[i];
        double bx = p->x[i + 1], by = p->y[i + 1];
        double seg_len = hypot(bx - ax, by - ay);
        double ahead = (1.0 - best_t) * seg_len;

        if (seg_len > 0.0 && remain <= ahead) {
            double t = best_t + remain / seg_len;
            *x_r = ax + t * (bx - ax);
            *y_r = ay + t * (by - ay);
            *th_r = atan2(by - ay, bx - ax);
            return;
        }
        remain -= ahead;
        i++;
        best_t = 0.0;
    }

    /* 3) 経路端を超えた: 終点にクランプし、向きは最終セグメントの向き */
    *x_r = p->x[p->n - 1];
    *y_r = p->y[p->n - 1];
    *th_r = atan2(p->y[p->n - 1] - p->y[p->n - 2],
                  p->x[p->n - 1] - p->x[p->n - 2]);
}

void path_tracking_error(double x, double y, double th,
                         double x_r, double y_r, double th_r,
                         double *x_e, double *y_e, double *th_e)
{
    double dx = x_r - x;
    double dy = y_r - y;
    *x_e = cos(th) * dx + sin(th) * dy;
    *y_e = -sin(th) * dx + cos(th) * dy;
    *th_e = path_normalize_angle(th_r - th);
}

int path_goal_crossed(const path_t *p, double x, double y)
{
    double gx = p->x[p->n - 1], gy = p->y[p->n - 1];
    double ax = p->x[p->n - 2], ay = p->y[p->n - 2];
    double seg_len = hypot(gx - ax, gy - ay);
    double ux, uy;

    if (seg_len <= 0.0) {
        return 0;
    }
    ux = (gx - ax) / seg_len;
    uy = (gy - ay) / seg_len;
    return ((x - gx) * ux + (y - gy) * uy >= 0.0) ? 1 : 0;
}

double path_goal_scaled_vr(double v_r, double goal_dist, double slow_radius)
{
    if (slow_radius <= 0.0 || goal_dist >= slow_radius) {
        return v_r;
    }
    return v_r * goal_dist / slow_radius;
}
```

```c
#include "model.h"

#include <math.h>

void model_error_dynamics(double v_r, double w_r, double ts,
                          double *A, double *B)
{
    /* A_c = [[0, w_r, 0], [-w_r, 0, v_r], [0, 0, 0]] */
    A[0] = 1.0;         A[1] = ts * w_r;   A[2] = 0.0;
    A[3] = -ts * w_r;   A[4] = 1.0;        A[5] = ts * v_r;
    A[6] = 0.0;         A[7] = 0.0;        A[8] = 1.0;

    /* B_c = [[-1, 0], [0, 0], [0, -1]] */
    B[0] = -ts;   B[1] = 0.0;
    B[2] = 0.0;   B[3] = 0.0;
    B[4] = 0.0;   B[5] = -ts;
}

void model_step(double *x, double *y, double *th, double v, double w, double dt)
{

       順序を変えると数値が一致しなくなる。 */
    *x += v * cos(*th) * dt;
    *y += v * sin(*th) * dt;
```

```c
static double naive_cost(const double *A, const double *B, int N,
                         const double *z0, const cost_weights_t *w,
                         const double *du_prev, const double *U, double lam)
{
    double z[MPC_NX];
    double zn[MPC_NX];
    double prev[MPC_NU];
    double J = 0.0;
    int k, i;

    la_copy(z, z0, MPC_NX);
    la_copy(prev, du_prev, MPC_NU);

    for (k = 0; k < N; k++) {
        const double *du = &U[k * MPC_NU];

        for (i = 0; i < MPC_NX; i++) {
            J += w->q[i] * z[i] * z[i];
        }
        for (i = 0; i < MPC_NU; i++) {
            if (w->use_r) {
                J += w->r[i] * du[i] * du[i];
            } else {
                J += lam * fabs(du[i]);
            }
            if (w->move_suppress > 0.0) {
                double d = du[i] - prev[i];
                J += w->move_suppress * d * d;
            }
        }
        /* z_{k+1} = A z_k + B du_k */
        la_matvec(A, z, zn, MPC_NX, MPC_NX);

            double Bu[MPC_NX];
            la_matvec(B, du, Bu, MPC_NX, MPC_NU);
            for (i = 0; i < MPC_NX; i++) {
                zn[i] += Bu[i];
            }
        }
        la_copy(z, zn, MPC_NX);
        la_copy(prev, du, MPC_NU);
    }
    for (i = 0; i < MPC_NX; i++) {
        J += w->qf_scale * w->q[i] * z[i] * z[i];
    }
    return J;
}

                             int nv, const double *U, double lam)
{
    double PU[MPC_NV_MAX];
    double J;
    int i;

    for (i = 0; i < nv; i++) {
        J += lam * fabs(U[i]);
    }
    return J;
}
```

```c
#ifndef SPARSE_MPC_CONDENSE_H
#define SPARSE_MPC_CONDENSE_H

#include "config.h"

/* 状態を消去して決定変数を入力偏差 U = [du_0; ...; du_{N-1}] だけにする。
   これにより KKT 行列の LDLT 分解が不要になり、全データが静的配列に収まる。
   代償として Gam が密になり、メモリ・演算とも O(N^2) になる。 */

typedef struct {
    int N;
    int nz;                                   /* 3*(N+1) */
    int nv;                                   /* 2*N     */
    double Phi[MPC_NZ_MAX * MPC_NX];          /* nz x 3   : ブロック行 k = A^k */
    double Gam[MPC_NZ_MAX * MPC_NV_MAX];      /* nz x nv  : ブロック (k,j) = A^{k-1-j} B */
} condense_t;

typedef struct {
    double q[MPC_NX];      /* 状態重みの対角 */
    double qf_scale;       /* 終端重みの倍率（Qf = qf_scale * Q） */
    double r[MPC_NU];      /* L2 の入力重み対角。use_r = 0 なら未使用 */
    int use_r;             /* 1 = L2, 0 = L1 */
    double move_suppress;  /* Δu 率ペナルティの係数 */
} cost_weights_t;

/* 0 = 成功, -1 = N が 1 未満または MPC_N_MAX 超過 */
int condense_build(condense_t *c, const double *A, const double *B, int N);

/* J = U^T P U + 2 q^T U + c0 (+ lam*||U||_1 は呼び出し側で加える)
   となる P (nv x nv), qv (nv), c0 を作る。
   c0 は Python 版の目的関数値と突き合わせるために必要。 */
void condense_cost(const condense_t *c, const double *z0,
                   const cost_weights_t *w, const double *du_prev,
                   double *P, double *qv, double *c0);

#endif /* SPARSE_MPC_CONDENSE_H */
```

```c
#include "condense.h"

#include "linalg.h"

int condense_build(condense_t *c, const double *A, const double *B, int N)
{
    /* AB[i] = A^i B。ブロック Toeplitz なので i ごとに一度作れば使い回せる。 */
    double AB[MPC_N_MAX * MPC_NX * MPC_NU];
    double Apow[MPC_NX * MPC_NX];
    double tmp[MPC_NX * MPC_NX];
    int nz, nv, k, j, i, l;

    if (N < 1 || N > MPC_N_MAX) {
        return -1;
    }
    nz = MPC_NX * (N + 1);
    nv = MPC_NU * N;
    c->N = N;
    c->nz = nz;
    c->nv = nv;

    /* Apow = I */
    la_zero(Apow, MPC_NX * MPC_NX);
    for (i = 0; i < MPC_NX; i++) {
        Apow[i * MPC_NX + i] = 1.0;
    }

    for (k = 0; k <= N; k++) {
        /* Phi のブロック行 k に A^k を書く */
        for (i = 0; i < MPC_NX; i++) {
            for (j = 0; j < MPC_NX; j++) {
                c->Phi[(k * MPC_NX + i) * MPC_NX + j] = Apow[i * MPC_NX + j];
            }
        }
        if (k < N) {
            la_matmul(Apow, B, &AB[k * MPC_NX * MPC_NU], MPC_NX, MPC_NX, MPC_NU);
            la_matmul(A, Apow, tmp, MPC_NX, MPC_NX, MPC_NX);
            la_copy(Apow, tmp, MPC_NX * MPC_NX);
        }
    }

    la_zero(c->Gam, nz * nv);
    for (k = 1; k <= N; k++) {
        for (j = 0; j < k; j++) {
            const double *blk = &AB[(k - 1 - j) * MPC_NX * MPC_NU];
            for (i = 0; i < MPC_NX; i++) {
                for (l = 0; l < MPC_NU; l++) {
                    c->Gam[(k * MPC_NX + i) * nv + (j * MPC_NU + l)] =
                        blk[i * MPC_NU + l];
                }
            }
        }
    }
    return 0;
}

void condense_cost(const condense_t *c, const double *z0,
                   const cost_weights_t *w, const double *du_prev,
                   double *P, double *qv, double *c0)
{
    double qt[MPC_NZ_MAX];      /* Q~ の対角 */
    double pz[MPC_NZ_MAX];      /* Phi z0 */
    int nz = c->nz, nv = c->nv, N = c->N;
    int r, i, j, k, l;
    double acc;

    /* Q~ = blkdiag(0, Q, ..., Q, Qf)。
       ブロック 0 を 0 にするのは z0 が定数で最適化に効かないため。
       その分は c0 に別途足す（Python の目的関数値と一致させるのに必要）。 */
    for (k = 0; k <= N; k++) {
        for (i = 0; i < MPC_NX; i++) {
            double val;
            if (k == 0) {
                val = 0.0;
            } else if (k == N) {
                val = w->qf_scale * w->q[i];
            } else {
                val = w->q[i];
            }
            qt[k * MPC_NX + i] = val;
        }
    }

    la_matvec(c->Phi, z0, pz, nz, MPC_NX);

    /* P = Gam^T diag(qt) Gam,  qv = Gam^T diag(qt) pz,  c0 = pz^T diag(qt) pz */
    la_zero(P, nv * nv);
    la_zero(qv, nv);
    acc = 0.0;
    for (r = 0; r < nz; r++) {
        const double *g = &c->Gam[r * nv];
        double wr = qt[r];
        if (wr == 0.0) {
            continue;
        }
        acc += wr * pz[r] * pz[r];
        for (i = 0; i < nv; i++) {
            double gi = g[i];
            if (gi == 0.0) {
                continue;
            }
            qv[i] += wr * gi * pz[r];
            for (j = 0; j < nv; j++) {
                P[i * nv + j] += wr * gi * g[j];
            }
        }
    }

    /* z0 の項は Python が含めているので c0 に足す */
    for (i = 0; i < MPC_NX; i++) {
        acc += w->q[i] * z0[i] * z0[i];
    }

    /* L2 の入力重み R~ */
    if (w->use_r) {
        for (k = 0; k < N; k++) {
            for (l = 0; l < MPC_NU; l++) {
                int idx = k * MPC_NU + l;
                P[idx * nv + idx] += w->r[l];
            }
        }
    }

    /* 移動抑制 ms * sum ||du_k - du_{k-1}||^2 （du_{-1} = du_prev）
       P には D^T D、qv には -ms*du_prev（ブロック 0）、c0 には ms*||du_prev||^2 */
    if (w->move_suppress > 0.0) {
        double ms = w->move_suppress;
        for (k = 0; k < N; k++) {
            for (l = 0; l < MPC_NU; l++) {
                int idx = k * MPC_NU + l;
                /* 自分の項に加え、次のステップの差分にも現れる（k = N-1 を除く） */
                P[idx * nv + idx] += (k <= N - 2) ? 2.0 * ms : ms;
                if (k >= 1) {
                    int prev = (k - 1) * MPC_NU + l;
                    P[idx * nv + prev] -= ms;
                    P[prev * nv + idx] -= ms;
                }
            }
        }
        for (l = 0; l < MPC_NU; l++) {
            qv[l] -= ms * du_prev[l];
            acc += ms * du_prev[l] * du_prev[l];
        }
    }

    *c0 = acc;
}
```

```c
#define SPARSE_MPC_FISTA_H

#include "config.h"

/* box 制約付き L1 正則化二次計画を近接勾配法（FISTA）で解く。

   min_U  U^T P U + 2 q^T U + lam*||U||_1   s.t.  lo <= U <= hi

   分解を使わないので必要なのは行列ベクトル積だけ。作業領域はすべてスタック上の
   固定長配列で、動的確保は行わない。 */

typedef struct {
    int max_iter;    /* 反復上限 */
    int fixed_iter;  /* 1 なら tol による早期終了をしない（ベンチの再現性用） */
    double tol;      /* 相対変化の停止閾値 */
} fista_opts_t;

typedef struct {
    int iters;       /* 実際に回った反復数 */
    double obj;      /* U^T P U + 2 q^T U + lam*||U||_1 （定数項 c0 は含まない） */
} fista_result_t;

void fista_opts_default(fista_opts_t *o);

/* U は入出力。前回解を入れておけばウォームスタートになる。
   box の外にあっても内側に射影してから開始する。 */
void fista_solve(const double *P, const double *qv, int nv,
                 double lam, const double *lo, const double *hi,
                 const fista_opts_t *opts, double *U, fista_result_t *res);

#endif /* SPARSE_MPC_FISTA_H */
```

```c
#include "fista.h"

#include "linalg.h"
#include "prox.h"

#include <math.h>

void fista_opts_default(fista_opts_t *o)
{
    o->max_iter = 200;
    o->fixed_iter = 0;
    o->tol = 1e-8;
}

/* P の最大固有値をべき乗法で推定する。P は対称半正定値なので
   ||P v|| / ||v|| が最大固有値に収束する。 */
static double power_max_eig(const double *P, int nv)
{
    double v[MPC_NV_MAX];
    double w[MPC_NV_MAX];
    double lam = 0.0;
    int i, it;

    /* 全成分同じ値で始めると支配的固有ベクトルと直交しうるので、
       決定的だが成分ごとに異なる初期値を使う。 */
    for (i = 0; i < nv; i++) {
        v[i] = 1.0 + 0.1 * (double)(i % 7);
    }
    for (it = 0; it < 50; it++) {
        double nrm;
        la_matvec(P, v, w, nv, nv);
        nrm = sqrt(la_dot(w, w, nv));
        if (nrm <= 1e-300) {
            return 0.0;
        }
        for (i = 0; i < nv; i++) {
            v[i] = w[i] / nrm;
        }
        lam = nrm;
    }
    return lam;
}

static double objective(const double *P, const double *qv, int nv,
                        double lam, const double *U)
{
    double PU[MPC_NV_MAX];
    double s;
    int i;

    la_matvec(P, U, PU, nv, nv);
    s = la_dot(U, PU, nv) + 2.0 * la_dot(qv, U, nv);
    if (lam > 0.0) {
        for (i = 0; i < nv; i++) {
            s += lam * fabs(U[i]);
        }
    }
    return s;
}

void fista_solve(const double *P, const double *qv, int nv,
                 double lam, const double *lo, const double *hi,
                 const fista_opts_t *opts, double *U, fista_result_t *res)
{
    double Y[MPC_NV_MAX];
    double Uprev[MPC_NV_MAX];
    double Unew[MPC_NV_MAX];
    double PY[MPC_NV_MAX];
    double step, Lc, t = 1.0;
    double obj_prev;
    int i, it;

    /* ウォームスタートが box の外にありうるので、まず内側に入れる */
    for (i = 0; i < nv; i++) {
        U[i] = prox_clip(U[i], lo[i], hi[i]);
    }

    /* 勾配は 2(PU + q) なのでリプシッツ定数は 2*lambda_max(P)。
       推定の誤差で発散しないよう 2% の余裕を持たせる。 */
    Lc = 2.0 * power_max_eig(P, nv) * 1.02;
    if (!(Lc > 0.0)) {
        Lc = 1.0;
    }
    step = 1.0 / Lc;

    la_copy(Uprev, U, nv);
    la_copy(Y, U, nv);
    obj_prev = objective(P, qv, nv, lam, U);

    for (it = 0; it < opts->max_iter; it++) {
        double t_new, beta, obj_new, diff = 0.0;

        la_matvec(P, Y, PY, nv, nv);
        for (i = 0; i < nv; i++) {
            double g = 2.0 * (PY[i] + qv[i]);
            Unew[i] = prox_l1_box(Y[i] - step * g, lam * step, lo[i], hi[i]);
        }

        t_new = 0.5 * (1.0 + sqrt(1.0 + 4.0 * t * t));
        beta = (t - 1.0) / t_new;

        /* FISTA は非単調。目的関数が悪化したら運動量を捨てて再起動する
           （これがないと L1 の解の近くで振動して収束が止まる）。
           そのための追加の行列ベクトル積は許容する。 */
        obj_new = objective(P, qv, nv, lam, Unew);
        if (obj_new > obj_prev) {
            t_new = 1.0;
            beta = 0.0;
        }

        for (i = 0; i < nv; i++) {
            double d = Unew[i] - Uprev[i];
            double a = fabs(d);
            if (a > diff) {
                diff = a;
            }
            Y[i] = Unew[i] + beta * d;
        }

        la_copy(Uprev, Unew, nv);
        t = t_new;
        obj_prev = obj_new;

        if (!opts->fixed_iter && diff <= opts->tol * (1.0 + la_norm_inf(Unew, nv))) {
            it++;
            break;
        }
    }

    la_copy(U, Uprev, nv);
    res->iters = it;
    res->obj = objective(P, qv, nv, lam, U);
}
```

```c
#include "mpc.h"

#include "linalg.h"
#include "model.h"
#include "prox.h"

void mpc_config_default(mpc_config_t *cfg)
{
    cfg->N = 15;
    cfg->ts = 0.1;
    cfg->reg = MPC_REG_L2;
    cfg->lam = 1.0;
    cfg->v_max = 0.15;
    cfg->w_max = 2.0;
    cfg->w.q[0] = 10.0;
    cfg->w.q[1] = 10.0;
    cfg->w.q[2] = 1.0;
    cfg->w.qf_scale = 10.0;
    cfg->w.r[0] = 1.0;
    cfg->w.r[1] = 1.0;
    cfg->w.use_r = 1;
    cfg->w.move_suppress = 0.0;
    fista_opts_default(&cfg->opts);
}

int mpc_init(mpc_t *m, const mpc_config_t *cfg)
{
    if (cfg->N < 1 || cfg->N > MPC_N_MAX) {
        return -1;
    }
    if (cfg->ts <= 0.0 || cfg->v_max <= 0.0 || cfg->w_max <= 0.0) {
        return -1;
    }
    m->cfg = *cfg;
    /* use_r は reg から一元的に導出する。呼び出し側の設定と食い違うと
       L1 のはずが R の二次項も入ってしまうため。 */
    m->cfg.w.use_r = (cfg->reg == MPC_REG_L2) ? 1 : 0;
    mpc_reset(m);
    return 0;
}

void mpc_reset(mpc_t *m)
{
    la_zero(m->U, MPC_NV_MAX);
    la_zero(m->du_prev, MPC_NU);
    la_zero(m->last_du0, MPC_NU);
    m->last_obj = 0.0;
    m->last_iters = 0;
    m->c0 = 0.0;
}

void mpc_set_du_prev(mpc_t *m, const double du[MPC_NU])
{
    la_copy(m->du_prev, du, MPC_NU);
}

int mpc_command(mpc_t *m, const double z0[MPC_NX], double v_r, double w_r,
                double *v, double *w)
{
    double A[MPC_NX * MPC_NX];
    double B[MPC_NX * MPC_NU];
    double u_nom[MPC_NU];
    double u_max[MPC_NU];
    fista_result_t res;
    double lam;
    int N = m->cfg.N;
    int nv = MPC_NU * N;
    int k, l;

    u_nom[0] = v_r;
    u_nom[1] = w_r;
    u_max[0] = m->cfg.v_max;
    u_max[1] = m->cfg.w_max;

    /* 実速度に課す box を du の box に写す: |u_nom + du| <= u_max */
    for (k = 0; k < N; k++) {
        for (l = 0; l < MPC_NU; l++) {
            m->lo[k * MPC_NU + l] = -u_max[l] - u_nom[l];
            m->hi[k * MPC_NU + l] = u_max[l] - u_nom[l];
        }
    }
    /* box が空でないことだけ確認する。u_max > 0 なので通常は必ず通るが、
       壊れた設定で黙って進まないための防御。 */
    if (!prox_box_valid(m->lo, m->hi, nv)) {
        return -1;
    }

    model_error_dynamics(v_r, w_r, m->cfg.ts, A, B);
    if (condense_build(&m->cond, A, B, N) != 0) {
        return -1;
    }
    condense_cost(&m->cond, z0, &m->cfg.w, m->du_prev,
                  m->P, m->qv, &m->c0);

    /* ウォームスタート: 前回解を1ステップ分シフトする（末尾は複製）。
       同じ問題を毎周期解き直すので、これだけで反復数がかなり減る。 */
    for (k = 0; k < N - 1; k++) {
        for (l = 0; l < MPC_NU; l++) {
            m->U[k * MPC_NU + l] = m->U[(k + 1) * MPC_NU + l];
        }
    }

    lam = (m->cfg.reg == MPC_REG_L1) ? m->cfg.lam : 0.0;
    fista_solve(m->P, m->qv, nv, lam, m->lo, m->hi, &m->cfg.opts, m->U, &res);

    m->last_du0[0] = m->U[0];
    m->last_du0[1] = m->U[1];
    m->last_obj = res.obj + m->c0;
    m->last_iters = res.iters;

    /* 次ステップの移動抑制のために、実際に採用した補正を覚える */
    m->du_prev[0] = m->U[0];
    m->du_prev[1] = m->U[1];

    *v = prox_clip(v_r + m->U[0], -m->cfg.v_max, m->cfg.v_max);
    *w = prox_clip(w_r + m->U[1], -m->cfg.w_max, m->cfg.w_max);
    return 0;
}
```

- [ ] **Step 1: `tools/gen_reference.py` を書く**

```python
#!/usr/bin/env python3
# -*- coding: utf-8 -*-
"""Python 版 MPC の解を参照データとして書き出す。

C 実装との数値一致検証（tests/test_reference.c）の真値になる。
生成には卒研リポジトリ（rover/mpc_core.py）と cvxpy が必要だが、
生成済み CSV をコミットするので C 側のテストは Python なしで走る。

MPCFollower.command() を使わないのは、それが OSQP を既定許容誤差
（eps_abs=eps_rel=1e-3）で呼ぶため。相対 1e-4 の真値には粗すぎるので、
パラメータを手で設定してから許容誤差を絞って解く。

使い方:
  cd ~/projects/sparse-optimal-control
  uv run python ~/projects/sparse-mpc-c/tools/gen_reference.py \
      --rover-dir ~/projects/sparse-optimal-control/rover \
      --out ~/projects/sparse-mpc-c/tests/data/reference.csv
"""

import argparse
import itertools
import os
import sys

import numpy as np


def main():
    ap = argparse.ArgumentParser()
    ap.add_argument("--rover-dir", required=True)
    ap.add_argument("--out", required=True)
    args = ap.parse_args()

    sys.path.insert(0, os.path.abspath(args.rover_dir))
    import cvxpy as cp
    from mpc_core import MPCFollower, error_dynamics

    z0_grid = [
        (0.00, 0.00, 0.00),
        (0.15, 0.00, 0.00),
        (0.10, 0.08, 0.20),
        (0.05, -0.12, -0.35),
        (-0.05, 0.20, 0.60),
        (0.20, -0.03, -0.10),
    ]
    vr_grid = [0.05, 0.10, 0.15]
    wr_grid = [-0.5, 0.0, 0.5]
    du_prev_grid = [(0.0, 0.0), (0.02, -0.30)]
    cfg_grid = [          # (reg, lam, move_suppress)
        ("l2", 1.0, 0.0),
        ("l2", 1.0, 2.0),
        ("l1", 0.25, 0.0),
        ("l1", 1.0, 0.0),
        ("l1", 2.0, 0.0),
        ("l1", 0.3, 2.0),
    ]
    n_grid = [5, 15]
    ts = 0.1

    rows = []
    skipped = 0
    for N, (reg, lam, ms) in itertools.product(n_grid, cfg_grid):
        mpc = MPCFollower(N=N, ts=ts, reg=reg, lam=lam,
                          v_max=0.15, w_max=2.0, move_suppress=ms)
        for z0, v_r, w_r, du_prev in itertools.product(
                z0_grid, vr_grid, wr_grid, du_prev_grid):
            # move_suppress=0 では du_prev が問題に効かないので1通りに絞る
            if ms == 0.0 and du_prev != (0.0, 0.0):
                continue

            mpc.z0.value = np.array(z0)
            mpc.u_nom.value = np.array([v_r, w_r])
            A, B = error_dynamics(v_r, w_r, ts)
            mpc.A.value = A
            mpc.B.value = B
            if mpc.du_prev is not None:
                mpc.du_prev.value = np.array(du_prev)

            try:
                mpc.prob.solve(solver=cp.OSQP, warm_start=False,
                               eps_abs=1e-10, eps_rel=1e-10,
                               max_iter=400000)
            except cp.error.SolverError:
                skipped += 1
                continue
            if mpc.prob.value is None or mpc.du.value is None:
                skipped += 1
                continue

            du0 = mpc.du.value[:, 0]
            rows.append((N, ts, 1 if reg == "l1" else 0, lam, ms,
                         v_r, w_r, z0[0], z0[1], z0[2],
                         du_prev[0], du_prev[1],
                         float(mpc.prob.value), float(du0[0]), float(du0[1])))

    with open(args.out, "w", encoding="utf-8") as f:
        f.write("N,ts,reg,lam,ms,v_r,w_r,x_e,y_e,th_e,"
                "dup_v,dup_w,obj,du0_v,du0_w\n")
        for r in rows:
            f.write("%d,%.17g,%d,%.17g,%.17g,%.17g,%.17g,%.17g,%.17g,%.17g,"
                    "%.17g,%.17g,%.17g,%.17g,%.17g\n" % r)

    print("wrote %d rows (skipped %d) to %s" % (len(rows), skipped, args.out),
          file=sys.stderr)


if __name__ == "__main__":
    main()
```

```c
#ifndef SPARSE_MPC_RNG_H
#define SPARSE_MPC_RNG_H

/* 再現可能な乱数。標準の rand() は実装依存で環境をまたぐと同じ列にならないため、
   シミュレーション結果を再現可能にする目的で自前に持つ。 */

typedef struct {
    unsigned int s;
    int has_spare;
    double spare;
} rng_t;

void rng_seed(rng_t *r, unsigned int seed);
double rng_uniform(rng_t *r);   /* (0, 1) */

#endif /* SPARSE_MPC_RNG_H */
```

`src/rng.c`:

```c
#include "rng.h"

#include <math.h>

/* M_PI は C99 の -pedantic では標準外なので自前で持つ */
static const double RNG_TWO_PI = 6.28318530717958647692528676655900577;

void rng_seed(rng_t *r, unsigned int seed)
{
    /* xorshift は状態 0 から抜け出せないので 0 を避ける */
    r->s = (seed == 0u) ? 0x9E3779B9u : seed;
    r->has_spare = 0;
    r->spare = 0.0;
}

static unsigned int next_u32(rng_t *r)
{
    unsigned int x = r->s;
    x ^= x << 13;
    x ^= x >> 17;
    x ^= x << 5;
    r->s = x;
    return x;
}

double rng_uniform(rng_t *r)
{
    /* 0 と 1 をちょうど取らないようにする（log(0) を避けるため） */
    return ((double)next_u32(r) + 0.5) / 4294967296.0;
}

double rng_gauss(rng_t *r)
{
    double u1, u2, mag;

    if (r->has_spare) {
        r->has_spare = 0;
        return r->spare;
    }
    u1 = rng_uniform(r);
    u2 = rng_uniform(r);
    mag = sqrt(-2.0 * log(u1));
    r->spare = mag * sin(RNG_TWO_PI * u2);
    r->has_spare = 1;
    return mag * cos(RNG_TWO_PI * u2);
}
```

```c
        bad = sc;
        bad.delay_steps = -1;
        TU_CHECK(sim_run(&g_mpc, &straight, 0.0, 0.0, 0.0, &bad, &r, NULL) == -1);
    }

    /* L1 は L2 より補正がスパースになる（Maximum Hands-off の本質） */

        mpc_config_default(&cfg);
        TU_CHECK(mpc_init(&g_mpc, &cfg) == 0);
```

```c
#include "sim.h"

#include "model.h"
#include "rng.h"

#include <math.h>

/* ω の符号判定の死帯。Python 版 sim_delay_probe.py の W_DEAD と同じ値。
   これがないと微小なノイズを符号反転として数えてしまう。 */
#define SIM_W_DEAD 0.05

void sim_config_default(sim_config_t *c)
{
    c->dt = 0.1;
    c->base_vr = 0.1;
    c->lookahead = 0.15;
    c->goal_tol = 0.05;
    c->slow_radius = 0.3;
    c->t_max = 120.0;
    c->delay_steps = 0;
    c->pos_noise = 0.0;
    c->yaw_noise = 0.0;
    c->seed = 1u;
}

int sim_run(mpc_t *m, const path_t *p, double x0, double y0, double th0,
            const sim_config_t *c, sim_result_t *out, FILE *trace)
{
    double buf_v[SIM_DELAY_MAX + 1];
    double buf_w[SIM_DELAY_MAX + 1];
    int head = 0, count = 0;
    int cap;
    rng_t rng;
    double x = x0, y = y0, th = th0;
    double t = 0.0;
    double cross_sq = 0.0;
    double sum_u = 0.0;
    int steps = 0, flips = 0, prev_sign = 0;
    int w_sat = 0, w_zero = 0, du_zero = 0;
    double max_abs_w = 0.0;
    double gx, gy, gd = 0.0;
    int reached = 0, solve_fail = 0;

    if (c->delay_steps < 0 || c->delay_steps > SIM_DELAY_MAX) {
        return -1;
    }
    if (c->dt <= 0.0 || p->n < 2) {
        return -1;
    }
    cap = c->delay_steps + 1;
    rng_seed(&rng, c->seed);

    gx = p->x[p->n - 1];
    gy = p->y[p->n - 1];

    while (t < c->t_max) {
        double xm, ym, thm;
        double x_r, y_r, th_r, x_e, y_e, th_e;
        double v_r, v_cmd, w_cmd, v_app, w_app;
        double z0[MPC_NX];
        int sign;

        gd = hypot(gx - x, gy - y);
        if (gd < c->goal_tol || path_goal_crossed(p, x, y)) {
            reached = 1;
            break;
        }

        /* 制御器が見るのはノイズの乗った測定値 */
        xm = x + ((c->pos_noise > 0.0) ? rng_gauss(&rng) * c->pos_noise : 0.0);
        ym = y + ((c->pos_noise > 0.0) ? rng_gauss(&rng) * c->pos_noise : 0.0);
        thm = th + ((c->yaw_noise > 0.0) ? rng_gauss(&rng) * c->yaw_noise : 0.0);

        path_reference_pose(p, xm, ym, c->lookahead, &x_r, &y_r, &th_r);
        path_tracking_error(xm, ym, thm, x_r, y_r, th_r, &x_e, &y_e, &th_e);
        v_r = path_goal_scaled_vr(c->base_vr, gd, c->slow_radius);

        z0[0] = x_e;
        z0[1] = y_e;
        z0[2] = th_e;
        if (mpc_command(m, z0, v_r, 0.0, &v_cmd, &w_cmd) != 0) {
            solve_fail = 1;
            break;
        }

        /* 入力遅延: delay_steps 前の指令を適用する。
           バッファが埋まるまでは直進巡航（Python 版と同じ扱い）。 */
        buf_v[(head + count) % cap] = v_cmd;
        buf_w[(head + count) % cap] = w_cmd;
        count++;
        if (count > c->delay_steps) {
            v_app = buf_v[head];
            w_app = buf_w[head];
            head = (head + 1) % cap;
            count--;
        } else {
            v_app = v_r;
            w_app = 0.0;
        }

        /* 指標（すべて実際に適用した指令で数える） */
        sum_u += (fabs(v_app) + fabs(w_app)) * c->dt;
        sign = (w_app > SIM_W_DEAD) ? 1 : ((w_app < -SIM_W_DEAD) ? -1 : 0);
        if (sign != 0) {
            if (prev_sign != 0 && sign != prev_sign) {
                flips++;
            }
            prev_sign = sign;
        }
        if (fabs(w_app) > max_abs_w) {
            max_abs_w = fabs(w_app);
        }
        if (fabs(w_app) > 1.5) {
            w_sat++;
        }
        if (fabs(w_app) < SIM_W_DEAD) {
            w_zero++;
        }
        if (fabs(m->last_du0[0]) < 1e-6 && fabs(m->last_du0[1]) < 1e-6) {
            du_zero++;
        }
        cross_sq += y_e * y_e;

        if (trace != NULL) {
            fprintf(trace,
                    "%.4f,%.6f,%.6f,%.6f,%.6f,%.6f,%.6f,%.6f,%.6f,%.6f,"
                    "%.6f,%.6f,%d\n",
                    t, x, y, th, x_e, y_e, th_e, v_r, v_app, w_app,
                    m->last_du0[0], m->last_du0[1], m->last_iters);
        }

        model_step(&x, &y, &th, v_app, w_app, c->dt);
        steps++;
        t += c->dt;
    }

    if (!reached && !solve_fail) {
        /* ループを抜けた時点で到達していることもある */
        gd = hypot(gx - x, gy - y);
        reached = (gd < c->goal_tol || path_goal_crossed(p, x, y)) ? 1 : 0;
    }

    out->reached = reached;
    out->solve_fail = solve_fail;
    out->steps = steps;
    out->t = t;
    out->rmse_cm = (steps > 0) ? 100.0 * sqrt(cross_sq / (double)steps) : 0.0;
    out->sum_u = sum_u;
    out->w_flips = flips;
    out->max_abs_w = max_abs_w;
    out->w_sat_ratio = (steps > 0) ? (double)w_sat / (double)steps : 0.0;
    out->w_zero_ratio = (steps > 0) ? (double)w_zero / (double)steps : 0.0;
    out->du_zero_ratio = (steps > 0) ? (double)du_zero / (double)steps : 0.0;
    return 0;
}
```

```c
    path_init(p, xs, ys, 3);

/* 指定条件で L字を1本走らせる */
static void run(sim_result_t *r, mpc_reg_t reg, double lam,
                double ms, int delay)
{
    mpc_config_t cfg;
    sim_config_t sc;
    path_t p;

    lturn(&p);
    sim_config_default(&sc);
    sc.delay_steps = delay;

    mpc_config_default(&cfg);
    cfg.reg = reg;
    cfg.lam = lam;
    cfg.w.move_suppress = ms;

    TU_CHECK(mpc_init(&g_mpc, &cfg) == 0);
    TU_CHECK(sim_run(&g_mpc, &p, 0.0, 0.0, 0.0, &sc, r, NULL) == 0);
    TU_CHECK(r->reached == 1);
}

int main(void)
{
    sim_result_t l2_d2, l1_d0, l1_d2, l1_ms_d2, l1_ms_d0;

    run(&l2_d2,    MPC_REG_L2, 1.0, 0.0, 2);
    run(&l1_d0,    MPC_REG_L1, 0.3, 0.0, 0);
    run(&l1_d2,    MPC_REG_L1, 0.3, 0.0, 2);
    run(&l1_ms_d2, MPC_REG_L1, 0.3, 2.0, 2);
    run(&l1_ms_d0, MPC_REG_L1, 0.3, 2.0, 0);

    printf("  L2      delay=2: 反転=%2d Σ|u|=%6.2f RMSE=%5.2fcm δu0率=%3.0f%%\n",
           l2_d2.w_flips, l2_d2.sum_u, l2_d2.rmse_cm, l2_d2.du_zero_ratio * 100.0);
    printf("  L1 ms=0 delay=0: 反転=%2d Σ|u|=%6.2f RMSE=%5.2fcm δu0率=%3.0f%%\n",
           l1_d0.w_flips, l1_d0.sum_u, l1_d0.rmse_cm, l1_d0.du_zero_ratio * 100.0);
    printf("  L1 ms=0 delay=2: 反転=%2d Σ|u|=%6.2f RMSE=%5.2fcm δu0率=%3.0f%%\n",
           l1_d2.w_flips, l1_d2.sum_u, l1_d2.rmse_cm, l1_d2.du_zero_ratio * 100.0);
    printf("  L1 ms=2 delay=2: 反転=%2d Σ|u|=%6.2f RMSE=%5.2fcm δu0率=%3.0f%%\n",
           l1_ms_d2.w_flips, l1_ms_d2.sum_u, l1_ms_d2.rmse_cm,
           l1_ms_d2.du_zero_ratio * 100.0);
    printf("  L1 ms=2 delay=0: 反転=%2d Σ|u|=%6.2f RMSE=%5.2fcm δu0率=%3.0f%%\n",
           l1_ms_d0.w_flips, l1_ms_d0.sum_u, l1_ms_d0.rmse_cm,
           l1_ms_d0.du_zero_ratio * 100.0);

    /* 主張 1: 遅延を入れると L1 の符号反転が増える（チャタの発生） */
    TU_CHECK(l1_d2.w_flips > l1_d0.w_flips);

    /* 主張 2: 同じ遅延で L1 は L2 より荒れる */
    TU_CHECK(l1_d2.w_flips > l2_d2.w_flips);

    /* 主張 3: move_suppress を入れると遅延下のチャタが減る（対策の効果） */
    TU_CHECK(l1_ms_d2.w_flips < l1_d2.w_flips);
    TU_CHECK(l1_ms_d2.sum_u < l1_d2.sum_u);

    /* 主張 4: move_suppress を入れてもスパース性は残る
       （チャタを消す代償に L1 の売りを失っていないこと） */
    TU_CHECK(l1_ms_d0.du_zero_ratio > 0.0);

    /* 主張 5: どの条件でもゴールには着く（対策が追従を壊していない） */
    TU_CHECK(l1_ms_d2.rmse_cm < l1_d2.rmse_cm * 2.0);

    TU_REPORT();
}
```

```markdown
# sparse-mpc-c

農業用ロボットの経路追従における**スパース最適制御（Maximum Hands-off Control）**の
モデル予測制御を、外部ライブラリに依存しない C99 で実装したシミュレータ。

卒業研究（MPC + L1 正則化による経路追従）で Python + cvxpy により得た結果を、
組込みを想定した制約（**動的メモリ確保なし・外部依存ゼロ・固定長配列のみ**）で
書き直したもの。

## 何を解いているか

参照軌道まわりで線形化した誤差ダイナミクスに対し、毎周期この問題を解く。

    min  Σ_{k=0}^{N-1} [ zₖᵀQzₖ + reg(δuₖ) + ms‖δuₖ − δu_{k−1}‖² ] + z_NᵀQ_f z_N
    s.t. z_{k+1} = A zₖ + B δuₖ,  |u_nom + δuₖ| ≤ u_max

- 状態 `z = (x_e, y_e, θ_e)` は Kanayama 誤差、入力 `δu` は参照速度からの補正
- `reg(δu) = δuᵀRδu`（L2）または `λ‖δu‖₁`（L1 = Maximum Hands-off）
- L1 は補正をゼロに張り付かせ、「巡航は維持しつつ補正を打つ瞬間だけ動かす」
  スパースな操舵になる

## アルゴリズム

**状態を消去（condensing）して、決定変数を入力偏差 `U ∈ R^{2N}` だけにする。**

    Z = Φ z₀ + Γ U
    J = UᵀPU + 2qᵀU + c0 + λ‖U‖₁,   lo ≤ U ≤ hi

これを**近接勾配法（FISTA）**で解く。各反復は

1. 勾配ステップ `U ← Y − (1/L)·∇f(Y)`
2. ソフトしきい値（L1 の prox）
3. box へのクリップ

だけで、**行列分解が一切要らない**。そのため作業領域はすべてコンパイル時に
サイズの決まる静的配列に収まり、`malloc` を1回も呼ばない。

### 設計上のトレードオフ

condensing は `Γ` が密になるため、メモリと演算が **O(N²)** になる。
`N ≤ 50` 程度では問題にならないが、`N` を大きく取るなら sparse 形式の
ADMM（OSQP と同じ構造）のほうが有利。ここでは「分解なし・確保なし・依存なし」を
優先してこの構成を選んだ。

`prox` を「ソフトしきい値 → クリップ」の合成で書けるのは、1次元の凸関数を区間上で
最小化した解が無制約の最小点を区間にクリップしたものに等しいからである。
区間が 0 を含むかどうかは関係しない。
```
